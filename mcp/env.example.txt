# ────────────────────────────────────────────────────────────────
# Arc MCP Server — Environment Variables
# Copy this file to .env and fill in the values.
# Never commit .env or any file containing real secrets.
# ────────────────────────────────────────────────────────────────

# ── Network ───────────────────────────────────────────────────────
# "testnet" (default, safe) or "mainnet" (real USDC — irreversible).
ARC_NETWORK=testnet

# ── Agent wallet mode ─────────────────────────────────────────────
# "unsigned" (default): server builds tx, human signs with MetaMask/cast.
#   No key material on the server. Recommended for most setups.
# "circle": Circle Developer-Controlled Wallets API signs the transaction.
#   Requires CIRCLE_API_KEY, ENTITY_SECRET, CIRCLE_WALLET_ID below.
# "circle_agent": Phase 1 — Circle Agent Wallets via the Circle CLI.
#   No key material on the server. Requires `circle wallet login` CLI session.
#   Preferred for AI agent flows: 2-of-2 MPC, user custody, spending policies.
AGENT_MODE=unsigned

# ── MCP transport ─────────────────────────────────────────────────
# "stdio": Claude Desktop / Cursor (local process, no HTTP).
# "http" : Remote agents over HTTP + SSE.
# "both" : Both (default — useful during development).
MCP_TRANSPORT=both
MCP_PORT=3456

# ── Circle credentials (required only for AGENT_MODE=circle) ──────
CIRCLE_API_KEY=
ENTITY_SECRET=
CIRCLE_WALLET_ID=

# ── Spending limits ───────────────────────────────────────────────
# Maximum USDC per single transfer or contract write.
PER_CALL_LIMIT_USDC=10
# Maximum cumulative USDC in a rolling 24-hour window.
DAILY_LIMIT_USDC=100

# ── Phase 0: Kill switch ──────────────────────────────────────────
# Set to "true" to instantly block ALL write tools (transfer_usdc,
# call_contract_write). Takes effect on the next request — no restart needed.
# Default: false
KILL_SWITCH=false

# ── Phase 0: Address allowlist ────────────────────────────────────
# Comma-separated list of 0x recipient addresses permitted for transfer_usdc.
# REQUIRE_ALLOWLIST=true (default): an empty ALLOWED_ADDRESSES rejects ALL
#   transfers — agents cannot move USDC until at least one address is listed.
# REQUIRE_ALLOWLIST=false: an empty list means "allow any address" (less safe).
# Example: ALLOWED_ADDRESSES=0xAbc...123,0xDef...456
ALLOWED_ADDRESSES=
REQUIRE_ALLOWLIST=true

# ── Phase 0: Contract allowlist ───────────────────────────────────
# Comma-separated list of 0x contract addresses permitted for call_contract_write.
# An empty list hard-rejects ALL contract writes.
# Example: ALLOWED_CONTRACTS=0x3600000000000000000000000000000000000000
ALLOWED_CONTRACTS=

# ── Phase 0: Selector allowlist (optional) ────────────────────────
# Comma-separated list of 4-byte function selectors (0x + 8 hex chars).
# When set, only these specific functions may be called via call_contract_write.
# Omit or leave empty to allow all functions on the allowed contracts.
# Example (ERC-20 transfer + approve): ALLOWED_SELECTORS=0xa9059cbb,0x095ea7b3
ALLOWED_SELECTORS=

# ── Phase 0: Operation binding ────────────────────────────────────
# When true (default), every write log entry includes a keccak256 binding hash
# of the exact payload (to|amount|data|nonce|timestamp). Only entries with a
# binding.hash count toward the daily rolling limit, preventing log tampering
# from bypassing the cap.
ENFORCE_OPERATION_BINDING=true

# ── Phase 1: Circle Agent Wallets (via Circle CLI) ────────────────
# Required only when using the circle_agent tools (create_agent_wallet,
# list_agent_wallets, get_agent_policy, set_agent_policy, transfer_usdc_agent).
#
# 1. Install the Circle CLI:  npm install -g @circle-fin/cli
# 2. Authenticate:            circle wallet login <email> --testnet
# 3. Complete the email OTP.
#
# CIRCLE_CLI_PATH: optional — override if `circle` is not on your PATH.
#   Example: CIRCLE_CLI_PATH=/usr/local/bin/circle
# CIRCLE_CLI_PATH=
#
# Important constraints (from Circle docs):
#   - Each account supports up to 5 agent wallets.
#   - Wallets cannot be deleted once created.
#   - Spending policies (set_agent_policy) are MAINNET ONLY.
#     Testnet wallets have no Circle-enforced policies; Phase 0 guards apply.
#   - Policy changes require a human to complete an email OTP.
#     The MCP server NEVER automates or bypasses OTP confirmation.
#   - CLI sessions expire after 28 days; re-run `circle wallet login` to renew.

# ── Phase 3: x402 / Nanopayments ─────────────────────────────────
# Required only when using pay_x402 and discover_x402_services tools.
# Uses the same Circle CLI session as Phase 1 (circle wallet login).
# No extra credentials needed beyond an authenticated CLI session.
#
# X402_MAX_PRICE_USDC: optional server-wide ceiling for pay_x402 calls.
# Individual tool calls also supply their own maxPriceUsdc; whichever is
# lower takes effect. Omit to rely solely on per-call limits.
#   Example: X402_MAX_PRICE_USDC=0.10
# X402_MAX_PRICE_USDC=
#
# X402_ALLOWED_ENDPOINTS: optional comma-separated endpoint URL prefixes.
# When set, pay_x402 only calls endpoints that start with one of these
# prefixes. Protects against prompt injection routing payments to an
# attacker-controlled endpoint. Leave empty to allow any https:// URL.
#   Example: X402_ALLOWED_ENDPOINTS=https://api.example.com,https://data.partner.io
# X402_ALLOWED_ENDPOINTS=
#
# Notes:
#   - pay_x402 always runs Phase 0 kill switch + per-call cap + daily limit.
#   - Wallet spending policies (Phase 1) apply as a second Circle-enforced layer
#     on mainnet. Set an x402Cap via set_agent_policy for tightest control.
#   - circle services pay is rate-limited by Circle CLI session.
#   - All paid calls are logged with a binding hash in tool-calls.jsonl.

# ── Phase 4: Control Plane ────────────────────────────────────────
# Client + grant data is stored as JSON files alongside tool-calls.jsonl.
# CLIENTS_DATA_DIR: directory for agent-clients.json, agent-grants.json,
#   and kill-switch.json. Defaults to the project root (/home/user/app).
# CLIENTS_DATA_DIR=

# ── Log path ──────────────────────────────────────────────────────
# Path to the JSONL file where tool calls are logged.
# Relative paths are resolved from the mcp/ directory.
# The dashboard reads this file via the API server (server/api.ts).
# Default: ../tool-calls.jsonl (project root)
# TOOL_LOG_PATH=../tool-calls.jsonl

# ── Phase 5: Authentication ───────────────────────────────────────
# API_AUTH_TOKEN: bearer token required on all MCP HTTP requests (except /health).
# When empty, the HTTP endpoint is unauthenticated — acceptable for local dev,
# REQUIRED for any network-accessible deployment.
# Generate: openssl rand -hex 32
# API_AUTH_TOKEN=

# ADMIN_AUTH_TOKEN: separate token for /admin/* routes (audit export, system status).
# Falls back to API_AUTH_TOKEN when not set.
# ADMIN_AUTH_TOKEN=

# DASHBOARD_API_TOKEN: bearer token for the dashboard API server (server/api.ts, port 3001).
# When empty, the dashboard API is unauthenticated. Set for production.
# DASHBOARD_API_TOKEN=

# ── Phase 5: Rate limiting ────────────────────────────────────────
# RATE_LIMIT_WINDOW_MS: sliding window duration in milliseconds (default 60000 = 1 min).
# RATE_LIMIT_MAX: max requests per window per IP on all routes (default 60).
# RATE_LIMIT_WRITE_MAX: max requests per window per IP on /sse and /message (default 20).
# Tuning: start conservative and raise only if legitimate agent traffic is throttled.
# RATE_LIMIT_WINDOW_MS=60000
# RATE_LIMIT_MAX=60
# RATE_LIMIT_WRITE_MAX=20

# ── Phase 5: Mainnet confirmation ────────────────────────────────
# The server warns on startup when ARC_NETWORK=mainnet. No additional flag is
# needed — the warning is always emitted and cannot be suppressed.
# See docs/MAINNET.md for the full mainnet readiness checklist.
