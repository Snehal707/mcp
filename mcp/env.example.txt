# ─────────────────────────────────────────────────────────────────────────────
# Arc MCP Server — Environment Variables
# Rename this file to .env and fill in your values. Never commit .env.
# ─────────────────────────────────────────────────────────────────────────────

# ── Wallet mode ───────────────────────────────────────────────────────────────
# "circle"   — Circle Developer-Controlled Wallets API signs transactions.
#              Requires CIRCLE_API_KEY + ENTITY_SECRET below.
#              Private keys never touch this server; Circle's infrastructure
#              holds them under your encrypted entity secret.
# "unsigned" — Server builds tx objects but does NOT sign or send.
#              Returns unsigned tx JSON for the human operator to sign in
#              MetaMask / Frame / cast. Zero key material on the server.
AGENT_MODE=unsigned

# ── MCP transport ─────────────────────────────────────────────────────────────
# "stdio"  — Local process transport (Claude Desktop, Cursor).
# "http"   — HTTP + SSE remote transport (ChatGPT plugins, remote agents).
# "both"   — Run both simultaneously (useful for dev).
MCP_TRANSPORT=both

# HTTP port (used when MCP_TRANSPORT is "http" or "both")
MCP_PORT=3456

# ── Circle credentials (required only when AGENT_MODE=circle) ─────────────────
# Get your API key from https://console.circle.com -> Developer -> API Keys
CIRCLE_API_KEY=

# 32-byte hex entity secret (without 0x prefix).
# Register once at: https://developers.circle.com/wallets/dev-controlled/register-entity-secret
# Store the recovery file offline. Losing it means permanent loss of wallet access.
ENTITY_SECRET=

# Wallet ID to use for outbound transactions (created once via the Circle API).
# Leave blank on first run; the server prints instructions to create one.
CIRCLE_WALLET_ID=

# ── Spending limits (amounts in USDC, human-readable) ─────────────────────────
# Maximum USDC per single transfer / write call.
PER_CALL_LIMIT_USDC=10

# Maximum USDC across ALL calls in a rolling 24-hour window.
DAILY_LIMIT_USDC=100

# Comma-separated 0x addresses the server may send USDC to.
# Leave blank to allow any address (NOT recommended for production).
# Example: ALLOWED_ADDRESSES=0xAbc123...,0xDef456...
ALLOWED_ADDRESSES=

# ── Arc network ───────────────────────────────────────────────────────────────
# "testnet" (default) or "mainnet". Mainnet moves real USDC — use with caution.
ARC_NETWORK=testnet

# ── Tool call log ─────────────────────────────────────────────────────────────
# Path to the JSONL file the server appends one record per tool invocation.
# The dashboard reads this file to render the live call log.
TOOL_LOG_PATH=../tool-calls.jsonl
